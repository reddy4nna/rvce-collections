//system1
% H(n) = 1 / (1 + 0.9z^-1)

b = [1 0];
a = [1 0.9];

/linearity
x1 = rand(1,10);
x2 = rand(1,10);

y = filter(b,a,2*x1 + 2*x2);

y1 = filter(b,a,x1);
y2 = filter(b,a,x2);

y_sum = 2*y1 + 2*y2;

% Checking linearity
if (y-y_sum==0)
    disp('System 1 is Linear');
else
    disp('System 1 is Non Linear');
end

//causality
n = -10:10;

% Unit impulse signal
x = [zeros(1,10) 1 zeros(1,10)];

% System response
h = filter(b,a,x);

subplot(2,1,1);
stem(n,h);

title('System 1');
xlabel('Samples');
ylabel('Amplitude');


//stability
s = sum(abs(h));

if s < 1000
    disp('System 1 is Stable');
else
    disp('System 1 is Unstable');
end


//system2
% y(n) = e^x

//linearity
x3 = rand(1,10);
x4 = rand(1,10);

% Individual outputs
y3 = exp(x3);
y4 = exp(x4);

% Combined outputs
left_side = 5*y3 + 5*y4;

% Output for combined input
right_side = exp(5*x3 + 5*x4);

% Checking linearity
if isequal(left_side,right_side)
    disp('System 2 is Linear');
else
    disp('System 2 is Non Linear');
end
